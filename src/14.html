/* clamp - (min, preferenssi, max) */ /* hyvä kun haluat rajoittaa dynaamisia arvoja */ h1 { font-size: clamp(1.5rem, 5vw, 3rem); } .container { width: clamp(300px, 60%, 1200px); margin-inline: auto; } img { /* rajoita maksimi kokoa */ width: min(100%, 600px); } .box { /* rajoita minimi kokoa */ width: max(250px, 30%); } /* calc - toimii eri yksikköjen välisiin laskuihin */ .container { width: calc(100% - 2rem); } /* round - pyöristä lähimpään 50px kertoimeen */ .box { width: round(100%, 50px); }
